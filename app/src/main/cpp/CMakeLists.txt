cmake_minimum_required(VERSION 3.22)
project(ink2tex LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
 set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
 set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
 set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
 set(GGML_CPU_ARM_ARCH "" CACHE STRING "" FORCE)
else()
 set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
endif()
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../vendor/llama.cpp llama)
add_subdirectory(../../../../vendor/llama.cpp/tools/mtmd mtmd)
if(ANDROID)
 add_library(ink2tex SHARED bridge.cpp engine.cpp cpu-projector.cpp)
 target_link_libraries(ink2tex PRIVATE llama mtmd android log)
 target_link_options(ink2tex PRIVATE "-Wl,-z,max-page-size=16384")
else()
 add_executable(ink2tex-host host.cpp engine.cpp)
 add_executable(ink2tex-projector-test projector-test.cpp cpu-projector.cpp)
 target_link_libraries(ink2tex-projector-test PRIVATE ggml-base)
 target_link_libraries(ink2tex-host PRIVATE llama mtmd)
 target_include_directories(ink2tex-host PRIVATE ../../../../vendor/llama.cpp/vendor/stb)
endif()
